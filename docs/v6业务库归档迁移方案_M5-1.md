```sql
-- ① 数量：公开读取面与脚本口径一致
SELECT (SELECT count(*) FROM mechanics) AS 词条,
       (SELECT count(*) FROM lineups) AS 手册根,
       (SELECT count(*) FROM lessons) AS 课程,
       (SELECT count(*) FROM scenarios WHERE status='PUBLISHED') AS 情境;
-- 期望：38 / 6 / 11 / 14
-- ② 已发布手册：每套恰有当前发布修订且 ≥1 公开引用（validatePublish 保证，抽查确认）
SELECT l.id, count(r.id) AS pub_revs,
       (SELECT count(*) FROM content_references cr
        WHERE cr.content_type='LINEUP' AND cr.content_id=l.id AND cr.usage_scope='PUBLIC') AS public_refs
FROM lineups l
LEFT JOIN lineup_revisions r ON r.lineup_id=l.id AND r.review_status='PUBLISHED' AND r.superseded_at IS NULL
GROUP BY l.id ORDER BY l.id;
-- 期望：每行 pub_revs=1；三分/日落海/大河/河洛/逐鹿 public_refs=3（两图+官网 RAW）、核心灌注流=3（英雄接口/官网指引/结构存证）
-- ③ 版本：v1.1.14 为 CURRENT
SELECT version_id, status FROM game_versions ORDER BY version_id;
```
